# ViralClip-AI dependencies
#
# System prerequisite (NOT installable via pip):
#   ffmpeg  -- must be on PATH. See README.md.

# ---- Video I/O ----
# moviepy 2.x moved VideoFileClip to the top-level package.
# src/transcriber.py uses `from moviepy import VideoFileClip`, which is 2.x-only.
moviepy>=2.0.0

# ---- Source download ----
# Keep this current: extractors break whenever YouTube changes.
yt-dlp>=2024.1.1

# ---- Speech-to-text ----
openai-whisper>=20231117

# ---- Speaker diarization: OPTIONAL, see requirements-diarize.txt ----

# without it. To enable:  pip install -r requirements-diarize.txt

# ---- Semantic transcript search ----

# ---- HTTP API (src/api/) ----
# Not needed for the CLI. Installed together for simplicity; split into
# requirements-api.txt if you want a slimmer CLI-only install.
fastapi>=0.110.0
uvicorn[standard]>=0.27.0
# Required by FastAPI to parse multipart/form-data, i.e. file uploads.
python-multipart>=0.0.9

# ---- Config / secrets ----
python-dotenv>=1.0.0

# ---- Pinned transitively, listed for clarity ----
# torch is pulled in by whisper, pyannote.audio and sentence-transformers.

# from https://pytorch.org/get-started/locally/ then run this file.
torch>=2.0.0
